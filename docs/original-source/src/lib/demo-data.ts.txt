export type DemoData = {
  festival?: FestivalInfo;

    latitude: number | null;

    latitude: number | null;

    id: string;
    stageId: string;
    artistId: string;

    typeof eventMapPoints.$inferSelect,
    "latitude" | "longitude"
  > & { latitude: number | null; longitude: number | null })[];
  announcements: (typeof eventAnnouncements.$inferSelect)[];
  savedPerformanceIds: string[];
  crew: typeof crews.$inferSelect;
  crewMembers: {
    userId: string;
    displayName: string;
    role: string;
    mode: string | null;
    mapX: number | null;
    mapY: number | null;
    zoneName: string | null;
    active: boolean | null;

  following: boolean;
};
export type OrganiserData = DemoData & {

      },
      recentActivity: [
        {

          action: "event.published",
          createdAt: eventDay("14:45"),
        },
      ],
    } as unknown as OrganiserData;
  }
  const [organiser] = await db
    .select()
    .from(organisers)
    .where(eq(organisers.id, DEMO_IDS.organiser));
  const [membership] = await db
    .select()
    .from(organiserMembers)

    .from(auditLogs)
    .where(eq(auditLogs.eventId, DEMO_IDS.event))

    .limit(12);

    ...eventData,
    organiser,
    membership,
    recentActivity,
  } as OrganiserData;
}
