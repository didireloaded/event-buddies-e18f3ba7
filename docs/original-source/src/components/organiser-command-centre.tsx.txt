function eventTime(value: Date | string) {
  return new Intl.DateTimeFormat("en-GB", {
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
    timeZone: "Africa/Windhoek",
  }).format(new Date(value));
}

function relativeActivity(value: Date | string) {
  const date = new Date(value);
  return `${date.toLocaleDateString("en-GB", { day: "2-digit", month: "short", timeZone: "Africa/Windhoek" })} · ${eventTime(date)}`;
}

async function action(body: Record<string, unknown>) {
  const response = await fetch("/api/organiser-actions", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const result = (await response.json()) as { ok: boolean; error?: string };
  if (!response.ok || !result.ok) throw new Error(result.error ?? "Action failed");
  return result;
}

export function OrganiserCommandCentre({ data }: Props) {
  const router = useRouter();
  const [workspace, setWorkspace] = useState<Workspace>("overview");
  const [mobileMenu, setMobileMenu] = useState(false);
  const [toast, setToast] = useState<string | null>(null);
  const [eventStatus, setEventStatus] = useState(data.event.status);
  const [performanceStates, setPerformanceStates] = useState<Record<string, string>>(
    Object.fromEntries(data.performances.map((performance) => [performance.id, performance.status])),
  );
  const [pending, setPending] = useState<string | null>(null);

  const notify = (message: string) => {
    setToast(message);
    window.setTimeout(() => setToast(null), 2600);
  };

  const changePerformance = async (performanceId: string, status: PerformanceStatus) => {
    const previous = performanceStates[performanceId];
    setPerformanceStates((current) => ({ ...current, [performanceId]: status }));
    setPending(performanceId);
    try {
      await action({ action: "set-performance-status", performanceId, status });
      notify(`Performance marked ${status.replace("_", " ")}`);
      router.refresh();
    } catch (error) {
      setPerformanceStates((current) => ({ ...current, [performanceId]: previous }));
      notify(error instanceof Error ? error.message : "Couldn’t update performance");
    } finally {
      setPending(null);
    }
  };

  const changeEventStatus = async (status: string) => {
    const sensitive = status === "cancelled" || status === "postponed";
    if (sensitive && !window.confirm(`Confirm marking Eternal Summer as ${status}? This affects every attendee.`)) return;
    const previous = eventStatus;
    setEventStatus(status);
    setPending("event-status");
    try {
      await action({ action: "set-event-status", status, confirmCritical: sensitive });
      notify(`Event status updated to ${status}`);
      router.refresh();
    } catch (error) {
      setEventStatus(previous);
      notify(error instanceof Error ? error.message : "Couldn’t update event status");
    } finally {
      setPending(null);
    }
  };

  return (
    <main className="organiser-shell">
      <OrganiserSidebar
        data={data}
        workspace={workspace}
        onWorkspace={(next) => { setWorkspace(next); setMobileMenu(false); }}
        mobileMenu={mobileMenu}
      />
      <section className="organiser-main">
        <OrganiserTopbar
          data={data}
          eventStatus={eventStatus}
          onEventStatus={changeEventStatus}
          pending={pending === "event-status"}
          onMenu={() => setMobileMenu(!mobileMenu)}
        />
        <div className="organiser-content">
          {workspace === "overview" && (
            <OverviewWorkspace
              data={data}
              performanceStates={performanceStates}
              onPerformance={changePerformance}
              onWorkspace={setWorkspace}
              pending={pending}
            />
          )}
          {workspace === "schedule" && (
            <ScheduleWorkspace
              data={data}
              performanceStates={performanceStates}
              onPerformance={changePerformance}
              pending={pending}
              notify={notify}
            />
          )}
          {workspace === "map" && <MapWorkspace data={data} notify={notify} />}
          {workspace === "communications" && <CommunicationsWorkspace data={data} notify={notify} />}
        </div>
      </section>
      {mobileMenu && <button className="organiser-scrim" aria-label="Close menu" onClick={() => setMobileMenu(false)} />}
      {toast && <div className="organiser-toast"><Check size={16} />{toast}</div>}
    </main>
  );
}

function CommunicationsWorkspace({ data, notify }: { data: OrganiserData; notify: (message: string) => void }) {
  const router = useRouter();
  const [title, setTitle] = useState("");
  const [body, setBody] = useState("");
  const [category, setCategory] = useState("general");
  const [priority, setPriority] = useState<NoticePriority>("normal");
  const [confirmed, setConfirmed] = useState(false);
  const [sending, setSending] = useState(false);
  const canSend = title.trim().length >= 3 && body.trim().length >= 5 && (priority !== "critical" || confirmed);

  const submit = async (event: FormEvent) => {
    event.preventDefault();
    if (!canSend) return;
    setSending(true);
    try {
      await action({ action: "publish-announcement", title, body, category, priority, confirmCritical: confirmed });
      setTitle(""); setBody(""); setPriority("normal"); setConfirmed(false);
      notify(priority === "critical" ? "Critical notice sent and pinned" : "Announcement published");
      router.refresh();
    } catch (error) {
      notify(error instanceof Error ? error.message : "Couldn’t publish announcement");
    } finally { setSending(false); }
  };

  const sorted = useMemo(() => [...data.announcements].sort((a, b) => new Date(b.publishAt).getTime() - new Date(a.publishAt).getTime()), [data.announcements]);
  return (
    <>
      <div className="workspace-heading"><div><span className="organiser-eyebrow">LIVE COMMUNICATIONS</span><h1>Announcements</h1><p>Reach followers and attendees already inside the event.</p></div></div>
      <div className="communications-layout">
        <form className="organiser-card announcement-composer" onSubmit={submit}>
          <div className="organiser-card-head"><span><small>COMPOSE</small><h2>New announcement</h2></span><MessageSquareText size={19} /></div>
          <label><span>Title <em>{title.length}/80</em></span><input value={title} onChange={(event) => setTitle(event.target.value.slice(0, 80))} placeholder="What do attendees need to know?" /></label>
          <label><span>Message <em>{body.length}/500</em></span><textarea value={body} onChange={(event) => setBody(event.target.value.slice(0, 500))} placeholder="Keep it concise and actionable." rows={5} /></label>
          <div className="composer-fields"><label><span>Category</span><select value={category} onChange={(event) => setCategory(event.target.value)}><option value="general">General</option><option value="schedule">Schedule</option><option value="safety">Safety</option><option value="transport">Transport</option></select></label><label><span>Priority</span><select value={priority} onChange={(event) => { setPriority(event.target.value as NoticePriority); setConfirmed(false); }}><option value="normal">Normal</option><option value="important">Important</option><option value="critical">Critical</option></select></label></div>
          <div className={`audience-preview ${priority}`}><span>{priority === "critical" ? <AlertTriangle size={18} /> : <Send size={18} />}</span><div><strong>{priority === "critical" ? "Critical delivery" : "Audience: everyone at this event"}</strong><small>{priority === "critical" ? "Push notification + pinned Event Mode notice" : "In-app update and enabled push subscribers"}</small></div></div>
          {priority === "critical" && <label className="critical-confirm"><input type="checkbox" checked={confirmed} onChange={(event) => setConfirmed(event.target.checked)} /><span><Check size={12} /></span><p><strong>I confirm this is a critical safety message.</strong><small>This action is permission-restricted and will be written to the audit log.</small></p></label>}
          <div className="composer-actions"><button type="button">Save draft</button><button type="submit" disabled={!canSend || sending}>{sending ? <Loader2 size={14} className="spin" /> : <Send size={14} />} {priority === "critical" ? "Send critical notice" : "Publish now"}</button></div>
        </form>
        <section className="organiser-card announcement-history"><div className="organiser-card-head"><span><small>PUBLISHED</small><h2>Recent announcements</h2></span><span className="announcement-count">{data.announcements.length}</span></div><div className="announcement-admin-list">{sorted.map((announcement) => <article key={announcement.id}><span className={announcement.priority}>{announcement.priority === "critical" ? <AlertTriangle size={14} /> : <Megaphone size={14} />}</span><div><small>{announcement.priority} · {announcement.category}</small><strong>{announcement.title}</strong><p>{announcement.body}</p><em>{relativeActivity(announcement.publishAt)}</em></div><button><MoreHorizontal size={16} /></button></article>)}</div></section>
      </div>
    </>
  );
}
