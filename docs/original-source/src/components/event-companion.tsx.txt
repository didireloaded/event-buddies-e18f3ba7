function useConnectivity() {
  const [online, setOnline] = useState(true);
  useEffect(() => {
    let active = true;
    const check = async () => {
      try {
        const response = await fetch("/manifest.webmanifest", {
          method: "HEAD",
          cache: "no-store",
          signal: AbortSignal.timeout(5000),
        });
        if (active) setOnline(response.ok);
      } catch {
        if (active) setOnline(false);
      }
    };
    const offline = () => setOnline(false);
    void check();
    window.addEventListener("online", check);
    window.addEventListener("offline", offline);
    window.addEventListener("focus", check);
    return () => {
      active = false;
      window.removeEventListener("online", check);
      window.removeEventListener("offline", offline);
      window.removeEventListener("focus", check);
    };
  }, []);
  return online;
}

function SheetDialog({
  title,
  children,
  onClose,
  returnFocus,
}: {
  title: string;
  children: ReactNode;
  onClose: () => void;
  returnFocus: () => HTMLElement | null;
}) {
  const dialog = useRef<HTMLDialogElement>(null);
  const restoreFocus = useRef(returnFocus);
  useEffect(() => {
    const element = dialog.current;
    const trigger = restoreFocus.current();
    element?.showModal();
    return () => {
      element?.close();
      requestAnimationFrame(() => {
        if (trigger?.isConnected) trigger.focus({ preventScroll: true });
      });
    };
  }, []);
  return (
    <dialog
      ref={dialog}
      className="eb-sheet"
      aria-labelledby="eb-sheet-title"
      onCancel={onClose}
      onClick={(event) => {
        if (event.target === event.currentTarget) onClose();
      }}
    >
      <div className="eb-sheet-inner">
        <div className="eb-sheet-handle" />
        <header>
          <h2 id="eb-sheet-title">{title}</h2>
          <button
            className="eb-round"
            aria-label="Close details"
            onClick={onClose}
          >
            <X size={21} />
          </button>
        </header>
        {children}
      </div>
    </dialog>
  );
}

function MeetingForm({
  eventStart,
  eventEnd,
  timezone,
  points,
  initial,
  selectedId,
  disabled,
  onSave,
}: {
  eventStart: Date;
  eventEnd: Date;
  timezone: string;
  points: Place[];
  initial: LocalMeeting | null;
  selectedId: string | null;
  disabled: boolean;
  onSave: (meeting: LocalMeeting | null) => Promise<void>;
}) {
  const [pointId, setPointId] = useState(
    initial?.pointId ?? selectedId ?? points[0]?.id ?? "",
  );
  const [at, setAt] = useState(
    initial ? initial.at.slice(0, 16) : localDateTime(eventStart, timezone),
  );
  return (
    <form
      className="eb-meeting-form"
      onSubmit={(event) => {
        event.preventDefault();
        void onSave({ pointId, at: `${at}:00+02:00` });
      }}
    >
      <p>Pick a landmark and a time. Your plan is saved on this device.</p>
      <label htmlFor="meeting-place">Meeting place</label>
      <select
        id="meeting-place"
        value={pointId}
        onChange={(event) => setPointId(event.target.value)}
        required
      >
        {points.map((point) => (
          <option key={point.id} value={point.id}>
            {point.name}
          </option>
        ))}
      </select>
      <label htmlFor="meeting-at">Date & time (CAT)</label>
      <input
        id="meeting-at"
        type="datetime-local"
        value={at}
        onChange={(event) => setAt(event.target.value)}
        min={localDateTime(eventStart, timezone)}
        max={localDateTime(eventEnd, timezone)}
        required
      />
      <button
        className="eb-button eb-button-pink"
        type="submit"
        disabled={disabled}
      >
        Save meeting plan <Check size={18} />
      </button>
      {initial && (
        <button
          className="eb-text-button"
          type="button"
          disabled={disabled}
          onClick={() => onSave(null)}
        >
          Remove meeting plan
        </button>
      )}
    </form>
  );
}

function MeetingCountdown({ meeting }: { meeting: LocalMeeting }) {
  const [now, setNow] = useState<number | null>(null);
  useEffect(() => {
    const timer = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(timer);
  }, []);
  const remaining =
    now === null
      ? null
      : Math.max(0, Math.ceil((Date.parse(meeting.at) - now) / 60000));
  return (
    <>
      <strong className="eb-meeting-time">
        {eventTime(meeting.at)} <span>CAT</span>
      </strong>
      <span className="eb-meeting-countdown">
        {remaining === null
          ? "Meeting time"
          : remaining === 0
            ? "Your meeting time has arrived"
            : remaining >= 1440
              ? `${Math.floor(remaining / 1440)} days to go`
              : `${Math.floor(remaining / 60)}h ${remaining % 60}m to go`}
      </span>
    </>
  );
}

function sortedCurrent(performances: Performance[], now: number | null) {
  if (now === null) return undefined;
  return performances.find(
    (performance) =>
      performance.status !== "cancelled" &&
      +new Date(performance.scheduledStart) <= now &&
      now < +new Date(performance.scheduledEnd),
  );
}

function matchesMapSearch(place: Place, query: string, data: DemoData) {
  const term = query.trim().toLowerCase();
  if (!term) return true;
  const artists =
    data.festival?.artists
      .filter((artist) => artist.stageId === place.id)
      .map((artist) => artist.name)
      .join(" ") ?? "";
  const aliases =
    place.type === "toilet"
      ? "bathroom restroom"
      : place.type === "first-aid"
        ? "medical medic help"
        : "";
  const stageAliases =
    place.id === "sos-main" ? "Savanna Savannah Main Stage" : "";
  return `${place.name} ${place.type} ${artists} ${aliases} ${stageAliases}`
    .toLowerCase()
    .includes(term);
}

function localDateTime(value: Date | string, timezone: string) {
  return new Intl.DateTimeFormat("sv-SE", {
    timeZone: timezone,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
  })
    .format(new Date(value))
    .replace(" ", "T");
}

function clusterMapPlaces(places: Place[], zoom: number) {
  const groups: { place: Place; ids: string[] }[] = [];
  for (const place of places) {
    const group = groups.find(
      (item) =>
        place.type !== "stage" &&
        item.place.type === place.type &&
        Math.hypot(item.place.x - place.x, item.place.y - place.y) < 14 / zoom,
    );
    if (group) group.ids.push(place.id);
    else groups.push({ place, ids: [place.id] });
  }
  return groups;
}
